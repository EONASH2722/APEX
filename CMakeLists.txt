cmake_minimum_required(VERSION 3.20)
project(ApexPhysicsCore VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(apex_physics Source/ApexPhysicsCore/ApexPhysics.cpp)
target_include_directories(apex_physics PUBLIC Source/ApexPhysicsCore/Public)
target_compile_options(apex_physics PRIVATE -Wall -Wextra -Wpedantic)
add_executable(apex_physics_tests Tests/PhysicsCoreTests.cpp)
target_link_libraries(apex_physics_tests PRIVATE apex_physics)
enable_testing()
add_test(NAME apex_physics_tests COMMAND apex_physics_tests)
